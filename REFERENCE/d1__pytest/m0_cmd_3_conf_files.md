# =============================================================================
# 27. КАК ПЕРЕОПРЕДЕЛИТЬ addopts ИЗ КОМАНДНОЙ СТРОКИ (например, убрать -s)
# =============================================================================
# В pytest.ini или pyproject.toml часто прописывают addopts = "-v -s".
# Иногда нужно временно отключить -s (захват вывода), не редактируя конфиг.

# ---------------------------------------------------------------------------
# Проблема: addopts всегда добавляется, и нет прямого "no-s".
# ---------------------------------------------------------------------------
# Параметры из addopts "склеиваются" с тем, что вы передали в командной строке,
# но если флаг конфликтует, обычно побеждает более поздний или более конкретный.
# У -s есть синонимы: -s == --capture=no
# Для отключения захвата используется --capture=fd (по умолчанию) или --capture=sys.

# ---------------------------------------------------------------------------
# Решение: передать --capture=fd в консоли
# ---------------------------------------------------------------------------
# Это переопределит поведение: вывод снова будет перехватываться,
# и вы не увидите print'ов в консоли во время выполнения тестов.
pytest --capture=fd

# Или, если нужно видеть вывод, но сохранить захват для отчётов:
pytest --capture=sys

# ---------------------------------------------------------------------------
# Проверка: как убедиться, что переопределение сработало?
# ---------------------------------------------------------------------------
# Можно добавить -v и посмотреть в выводе pytest строку:
# "rootdir: ..., configfile: pytest.ini"
# "addopts: -v -s"
# "cmdline: --capture=fd"
# Итоговое поведение будет с захватом вывода.

# ---------------------------------------------------------------------------
# Общий принцип: для любых конфликтующих опций
# ---------------------------------------------------------------------------
# Командная строка имеет приоритет над addopts.
# Если флаг не имеет противоположного, используйте альтернативный параметр,
# который меняет поведение (как --capture=fd против -s).
# Для логических флагов, у которых есть --no-версия (например, --no-header),
# можно использовать их (например, pytest --no-header).

# =============================================================================
# 28. КАК РАЗДЕЛИТЬ ПАРАМЕТРЫ ДЛЯ PYCHARM И КОНСОЛИ (-s только в IDE)
# =============================================================================
# Проблема: в pytest.ini/pyproject.toml прописано addopts = "-v -s".
# Вы хотите, чтобы -s действовал только при запуске из PyCharm,
# а при запуске из .cmd или обычной консоли вывод захватывался (без -s).
# Ниже три способа – от простого к гибкому.

# ---------------------------------------------------------------------------
# 28.1 ПРОСТОЙ: убрать -s из addopts, добавить его только в PyCharm  ---- У МЕНЯ НЕ СРАБОТАЛО!!!!!
# ---------------------------------------------------------------------------
# В pytest.ini или pyproject.toml оставить:
# addopts = -v
# А в PyCharm: Run -> Edit Configurations -> Additional Arguments = -s
# Тогда .cmd и консоль будут запускать pytest без -s.
# Это самое надёжное и прозрачное решение.

# ---------------------------------------------------------------------------
# 28.2 СРЕДНИЙ: использовать переменную окружения для автоматического -s в PyCharm
# ---------------------------------------------------------------------------
# В pytest.ini оставить addopts = -v, а в conftest.py добавить хук,
# который добавляет -s, если установлена переменная PYCHARM=1.
# В PyCharm в настройках конфигурации добавить Environment variable: PYCHARM=1

# conftest.py:
import os
import sys

def pytest_configure(config):
    if os.environ.get("PYCHARM") == "1":
        # Добавляем -s к аргументам, если его ещё нет
        if "-s" not in sys.argv and "--capture=no" not in sys.argv:
            sys.argv.append("-s")

# Теперь при запуске из PyCharm (с установленной переменной) будет -s,
# а при запуске из консоли или .cmd без этой переменной — без -s.

# ---------------------------------------------------------------------------
# 28.3 ГИБКИЙ: отдельные конфигурационные файлы для разных сред
# ---------------------------------------------------------------------------
# Можно создать pytest_ide.ini с содержимым:
# [pytest]
# addopts = -v -s
# А в консольном запуске указывать:
# pytest -c pytest_console.ini
# или в .cmd файле явно писать pytest -c pytest_console.ini
# Но это требует дублирования настроек, поэтому вариант 1 предпочтительнее.

# ---------------------------------------------------------------------------
# 28.4 ЧТО ДЕЛАТЬ С .CMD (Windows batch)
# ---------------------------------------------------------------------------
# В .cmd файле вы уже используете --capture=fd, что явно переопределяет -s.
# Это работает даже если addopts содержит -s, потому что --capture=fd
# перекрывает -s (который эквивалентен --capture=no). 
# Так что вариант с addopts = "-v -s" и .cmd с --capture=fd рабочий,
# но не очень аккуратный: в глобальном конфиге -s, а в .cmd вы его отменяете.
# Лучше всё-таки убрать -s из addopts, а в PyCharm добавить его явно.

# Итог:
# - Убрать -s из глобального addopts (pytest.ini или pyproject.toml).
# - В PyCharm в Run Configuration добавить Additional Arguments = -s.
# - В .cmd файлах ничего не менять (захват по умолчанию будет активен).
